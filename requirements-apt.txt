# Runtime dependencies, installed from Debian (trixie) apt — not pip.
#
#   sudo apt install python3-flask python3-flask-sqlalchemy python3-flask-login \
#     python3-flaskext.wtf python3-wtforms python3-email-validator \
#     python3-flask-talisman python3-flask-limiter python3-flask-babel \
#     python3-pil python3-qrcode \
#     python3-gunicorn fonts-dejavu-core python3-cairosvg \
#     python3-requests python3-jwt python3-cryptography
#
# Package name            Provides / used for
# ----------------------  -------------------------------------------------
# python3-flask           Flask web framework
# python3-flask-sqlalchemy  ORM session + teardown wiring
# python3-flask-login     admin session management
# python3-flaskext.wtf    Flask-WTF (module: flask_wtf) — forms + CSRF
# python3-wtforms         form fields / validators
# python3-email-validator e-mail address validation for forms
# python3-flask-talisman  security response headers / CSP
# python3-flask-limiter   login rate limiting
# python3-flask-babel     gettext translations + locale selection (module:
#                         flask_babel); pulls in python3-babel / pybabel
# python3-pil             Pillow — image validation, composing, PNG baking
# python3-qrcode          QR code image for assertion URLs
# python3-gunicorn        WSGI application server (run as: python3 -m gunicorn)
# fonts-dejavu-core       font used for composed badge titles
# python3-cairosvg        rasterise SVG uploads (optional — without it, SVG
#                         uploads are rejected with a clear message)
# python3-requests        outbound fetches for the /verify badge checker
# python3-jwt             verify signed (JWS) badges
# python3-cryptography    load issuer public keys for signed-badge verification
#
# Standard library only: smtplib/email (SMTP), hashlib (recipient hashing),
# werkzeug.security (password hashing).
#
# Optional, for running the test suite:
#   sudo apt install python3-pytest
#
# Translations: the compiled catalogs (badgeserver/translations/*/LC_MESSAGES/
# messages.mo) are committed, so a deployment needs nothing extra. To change a
# user-facing string you also need pybabel (python3-babel, a dependency of
# python3-flask-babel) — see contrib/i18n.sh and Docs/operations.md.
